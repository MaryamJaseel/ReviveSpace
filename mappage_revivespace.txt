import { useEffect, useState, useRef } from 'react';
import { useNavigate } from 'react-router-dom';
import { supabase } from '@/db/supabase';
import MainLayout from '@/components/layouts/MainLayout';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Card, CardContent } from '@/components/ui/card';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Skeleton } from '@/components/ui/skeleton';
import { MapPin, Filter, X, Building2, ArrowRight } from 'lucide-react';
import { STATUS_LABELS, type Property, type PropertyStatus, type PropertyCondition, type PropertyType } from '@/types/types';
import { getStatusColor, getStatusDotColor, getImpactScoreColor, formatDate } from '@/lib/utils-app';
import { MapContainer, TileLayer, CircleMarker, Popup, useMap } from 'react-leaflet';
import 'leaflet/dist/leaflet.css';

function MapBoundsUpdater({ properties }: { properties: Property[] }) {
  const map = useMap();
  useEffect(() => {
    if (properties.length > 0) {
      const bounds = properties.map(p => [p.latitude, p.longitude] as [number, number]);
      map.fitBounds(bounds, { padding: [40, 40] });
    }
  }, [properties, map]);
  return null;
}

export default function MapPage() {
  const navigate = useNavigate();
  const [properties, setProperties] = useState<Property[]>([]);
  const [filtered, setFiltered] = useState<Property[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedProperty, setSelectedProperty] = useState<Property | null>(null);
  const [filters, setFilters] = useState({
    status: 'all',
    type: 'all',
    condition: 'all',
  });
  const [showFilters, setShowFilters] = useState(false);
  const hasFit = useRef(false);

  useEffect(() => {
    supabase
      .from('properties')
      .select('*, property_images(id, image_url), property_votes(id)')
      .order('created_at', { ascending: false })
      .then(({ data }) => {
        const list = Array.isArray(data) ? data : [];
        setProperties(list);
        setFiltered(list);
        setLoading(false);
      });
  }, []);

  useEffect(() => {
    let result = [...properties];
    if (filters.status !== 'all') result = result.filter(p => p.status === filters.status);
    if (filters.type !== 'all') result = result.filter(p => p.property_type === filters.type);
    if (filters.condition !== 'all') result = result.filter(p => p.condition === filters.condition);
    setFiltered(result);
  }, [filters, properties]);

  const clearFilters = () => setFilters({ status: 'all', type: 'all', condition: 'all' });
  const hasFilters = filters.status !== 'all' || filters.type !== 'all' || filters.condition !== 'all';

  const defaultCenter: [number, number] = [40.72, -73.99];

  return (
    <MainLayout>
      <div className="flex flex-col h-[calc(100vh-3.5rem)] md:flex-row">
        {/* Left: Map */}
        <div className="flex-1 min-w-0 relative">
          {/* Filter Bar */}
          <div className="absolute top-3 left-3 right-3 z-[1000] flex items-center gap-2">
            <div className="flex-1 min-w-0 bg-background border border-border rounded shadow-md p-2 flex items-center gap-2 flex-wrap">
              <Button
                variant="ghost"
                size="sm"
                className="gap-1.5 text-xs h-8 shrink-0"
                onClick={() => setShowFilters(v => !v)}
              >
                <Filter className="w-3.5 h-3.5" />
                Filters
                {hasFilters && <Badge className="bg-primary text-primary-foreground text-xs px-1.5 h-4 border-0">{Object.values(filters).filter(v => v !== 'all').length}</Badge>}
              </Button>

              {showFilters && (
                <>
                  <Select value={filters.status} onValueChange={v => setFilters(f => ({ ...f, status: v }))}>
                    <SelectTrigger className="h-8 text-xs w-40 shrink-0">
                      <SelectValue placeholder="Status" />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="all">All Statuses</SelectItem>
                      {(Object.keys(STATUS_LABELS) as PropertyStatus[]).map(s => (
                        <SelectItem key={s} value={s}>{STATUS_LABELS[s]}</SelectItem>
                      ))}
                    </SelectContent>
                  </Select>

                  <Select value={filters.type} onValueChange={v => setFilters(f => ({ ...f, type: v }))}>
                    <SelectTrigger className="h-8 text-xs w-36 shrink-0">
                      <SelectValue placeholder="Type" />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="all">All Types</SelectItem>
                      <SelectItem value="residential">Residential</SelectItem>
                      <SelectItem value="commercial">Commercial</SelectItem>
                    </SelectContent>
                  </Select>

                  <Select value={filters.condition} onValueChange={v => setFilters(f => ({ ...f, condition: v }))}>
                    <SelectTrigger className="h-8 text-xs w-32 shrink-0">
                      <SelectValue placeholder="Condition" />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="all">All Conditions</SelectItem>
                      <SelectItem value="poor">Poor</SelectItem>
                      <SelectItem value="fair">Fair</SelectItem>
                      <SelectItem value="good">Good</SelectItem>
                    </SelectContent>
                  </Select>

                  {hasFilters && (
                    <Button variant="ghost" size="sm" className="h-8 text-xs gap-1 shrink-0" onClick={clearFilters}>
                      <X className="w-3 h-3" /> Clear
                    </Button>
                  )}
                </>
              )}

              <span className="text-xs text-muted-foreground ml-auto shrink-0">
                {filtered.length} propert{filtered.length !== 1 ? 'ies' : 'y'}
              </span>
            </div>
          </div>

          {/* Map Legend */}
          <div className="absolute bottom-6 left-3 z-[1000] bg-background border border-border rounded shadow-md p-2">
            <p className="text-xs font-semibold text-foreground mb-1.5">Status Legend</p>
            <div className="flex flex-col gap-1">
              {[
                { status: 'reported' as PropertyStatus, label: 'Reported' },
                { status: 'under_verification' as PropertyStatus, label: 'Under Verification' },
                { status: 'verified_vacant' as PropertyStatus, label: 'Verified' },
                { status: 'action_recommended' as PropertyStatus, label: 'Action Recommended' },
              ].map(({ status, label }) => (
                <div key={status} className="flex items-center gap-1.5">
                  <div className="w-3 h-3 rounded-full shrink-0" style={{ backgroundColor: getStatusDotColor(status) }} />
                  <span className="text-xs text-muted-foreground whitespace-nowrap">{label}</span>
                </div>
              ))}
            </div>
          </div>

          {loading ? (
            <Skeleton className="w-full h-full bg-muted" />
          ) : (
            <MapContainer
              center={defaultCenter}
              zoom={13}
              className="w-full h-full"
              zoomControl={true}
            >
              <TileLayer
                attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
                url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
              />
              {properties.length > 0 && !hasFit.current && (
                <MapBoundsUpdater properties={filtered} />
              )}
              {filtered.map(property => (
                <CircleMarker
                  key={property.id}
                  center={[property.latitude, property.longitude]}
                  radius={property.impact_score >= 80 ? 10 : property.impact_score >= 50 ? 8 : 6}
                  pathOptions={{
                    color: getStatusDotColor(property.status),
                    fillColor: getStatusDotColor(property.status),
                    fillOpacity: 0.8,
                    weight: 2,
                  }}
                  eventHandlers={{
                    click: () => setSelectedProperty(property),
                  }}
                >
                  <Popup>
                    <div className="min-w-[180px]">
                      <p className="font-semibold text-sm mb-1">{property.address}</p>
                      <Badge className={`text-xs border ${getStatusColor(property.status)} mb-2`}>
                        {STATUS_LABELS[property.status]}
                      </Badge>
                      <div className="text-xs text-gray-600 space-y-0.5">
                        <div>Type: <span className="capitalize">{property.property_type}</span></div>
                        <div>Condition: <span className="capitalize">{property.condition}</span></div>
                        <div>Impact Score: <span className="font-semibold">{property.impact_score}</span></div>
                      </div>
                      <button
                        onClick={() => navigate(`/property/${property.id}`)}
                        className="mt-2 text-xs text-blue-600 hover:underline font-medium"
                      >
                        View Details →
                      </button>
                    </div>
                  </Popup>
                </CircleMarker>
              ))}
            </MapContainer>
          )}
        </div>

        {/* Right: Property List / Detail Panel */}
        <div className="md:w-80 lg:w-96 shrink-0 border-l border-border overflow-y-auto bg-background">
          {selectedProperty ? (
            <div className="p-4">
              <div className="flex items-center justify-between mb-3 gap-2">
                <h3 className="font-semibold text-foreground text-balance">Property Detail</h3>
                <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={() => setSelectedProperty(null)}>
                  <X className="w-4 h-4" />
                </Button>
              </div>
              <PropertyDetailPanel property={selectedProperty} onNavigate={() => navigate(`/property/${selectedProperty.id}`)} />
            </div>
          ) : (
            <div className="p-4">
              <div className="flex items-center gap-2 mb-4">
                <MapPin className="w-4 h-4 text-primary shrink-0" />
                <h3 className="font-semibold text-foreground">
                  {filtered.length} Propert{filtered.length !== 1 ? 'ies' : 'y'}
                </h3>
              </div>
              {loading ? (
                <div className="space-y-3">
                  {[...Array(5)].map((_, i) => <Skeleton key={i} className="h-20 bg-muted" />)}
                </div>
              ) : filtered.length === 0 ? (
                <div className="text-center py-10">
                  <Building2 className="w-8 h-8 text-muted-foreground mx-auto mb-2" />
                  <p className="text-sm text-muted-foreground">No properties match your filters.</p>
                  {hasFilters && <Button variant="ghost" size="sm" className="mt-2" onClick={clearFilters}>Clear Filters</Button>}
                </div>
              ) : (
                <div className="space-y-2">
                  {filtered.map(p => (
                    <div
                      key={p.id}
                      onClick={() => setSelectedProperty(p)}
                      className="p-3 rounded border border-border hover:border-primary/40 cursor-pointer transition-colors"
                    >
                      <div className="flex items-start justify-between gap-2">
                        <Badge className={`text-xs border ${getStatusColor(p.status)} shrink-0`}>
                          {STATUS_LABELS[p.status]}
                        </Badge>
                        <span className={`text-xs font-bold shrink-0 ${getImpactScoreColor(p.impact_score)}`}>
                          {p.impact_score}
                        </span>
                      </div>
                      <p className="text-sm font-medium text-foreground mt-1.5 line-clamp-2">{p.address}</p>
                      <p className="text-xs text-muted-foreground mt-0.5 capitalize">{p.property_type} · {p.condition}</p>
                    </div>
                  ))}
                </div>
              )}
            </div>
          )}
        </div>
      </div>
    </MainLayout>
  );
}

function PropertyDetailPanel({ property, onNavigate }: { property: Property; onNavigate: () => void }) {
  return (
    <Card className="border border-border">
      <CardContent className="p-4 space-y-3">
        <Badge className={`text-xs border ${getStatusColor(property.status)}`}>
          {STATUS_LABELS[property.status]}
        </Badge>
        <div>
          <p className="font-semibold text-foreground text-balance">{property.address}</p>
          <p className="text-xs text-muted-foreground mt-0.5 capitalize">
            {property.property_type} · Condition: {property.condition}
          </p>
        </div>
        <div className="grid grid-cols-2 gap-2 text-xs">
          <div className="bg-muted rounded p-2">
            <p className="text-muted-foreground">Impact Score</p>
            <p className={`font-bold text-lg ${getImpactScoreColor(property.impact_score)}`}>{property.impact_score}</p>
          </div>
          <div className="bg-muted rounded p-2">
            <p className="text-muted-foreground">AI Confidence</p>
            <p className="font-bold text-lg text-foreground">{property.ai_confidence_score ?? 'N/A'}</p>
          </div>
        </div>
        {property.years_unused && (
          <p className="text-xs text-muted-foreground">Unused for ~{property.years_unused} year{property.years_unused !== 1 ? 's' : ''}</p>
        )}
        <p className="text-xs text-muted-foreground">Reported: {formatDate(property.created_at)}</p>
        <Button className="w-full gap-2" size="sm" onClick={onNavigate}>
          View Full Details <ArrowRight className="w-3.5 h-3.5" />
        </Button>
      </CardContent>
    </Card>
  );
}
